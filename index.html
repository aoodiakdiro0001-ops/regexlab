function enc1(s,kind){
if(kind==='url1')return encodeURIComponent(s);
if(kind==='url2')return encodeURIComponent(encodeURIComponent(s));
if(kind==='urldec'){try{return decodeURIComponent(s);}catch(e){return s;}}
if(kind==='b64e')return btoa(unescape(encodeURIComponent(s)));
if(kind==='b64d'){try{return decodeURIComponent(escape(atob(s.trim())));}catch(e){return s;}}
if(kind==='hex')return [...unescape(encodeURIComponent(s))].map(c=>c.charCodeAt(0).toString(16).padStart(2,'0')).join('');
if(kind==='html')return s.replace(/&/g,'&').replace(//g,'>').replace(/"/g,'"');
if(kind==='uni')return [...s].map(c=>'\\u'+c.charCodeAt(0).toString(16).padStart(4,'0')).join('');
if(kind==='fcc')return 'String.fromCharCode('+[...unescape(encodeURIComponent(s))].map(c=>c.charCodeAt(0)).join(',')+')';
if(kind==='upper')return s.toUpperCase();
if(kind==='mix')return [...s].map((c,i)=>i%2?c.toUpperCase():c.toLowerCase()).join('');
return s;
}
